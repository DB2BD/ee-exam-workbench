\[
I_{a1}=\frac{1.03}{j0.50}=-j2.06\,\mathrm{pu},
\qquad
|I_f^{(L-L)}|=\sqrt3|I_{a1}|=3.568\,\mathrm{pu}.
\]

I_{base}=\frac{1000\times10^6}{\sqrt3(500\times10^3)}=1154.7\,\mathrm{A},
\qquad |I_f|\approx4.120\,\mathrm{kA}.
\]
